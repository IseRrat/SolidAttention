cmake_minimum_required(VERSION 3.22)
project(SolidAttention VERSION 0.1.0 LANGUAGES C CXX)

option(SOLID_BUILD_TESTS "Build deterministic unit checks" ON)
option(SOLID_BUILD_RUNNER "Build the llama.cpp baseline runner" ON)
option(SOLID_CUDA "Build the CUDA backend" OFF)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

include(cmake/Upstream.cmake)
include(cmake/BuildInfo.cmake)
add_library(solid_core STATIC src/kv_store.cpp src/attention.cpp src/session.cpp)
target_include_directories(solid_core PUBLIC include)
target_include_directories(solid_core PRIVATE third_party/llama.cpp/vendor)
if(NOT MSVC)
    target_compile_options(solid_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_library(solid_config STATIC src/config.cpp src/filehash.cpp third_party/llama.cpp/vendor/hash/sha256/sha256.c)
target_include_directories(solid_config PUBLIC include PRIVATE third_party/llama.cpp/vendor
    third_party/llama.cpp/vendor/hash third_party/llama.cpp/vendor/hash/sha256)
if(MSVC)
    target_compile_options(solid_config PRIVATE /W4 /utf-8)
else()
    target_compile_options(solid_config PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

if(SOLID_BUILD_RUNNER)
    set(GGML_CUDA ${SOLID_CUDA} CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    add_subdirectory(third_party/llama.cpp EXCLUDE_FROM_ALL)
    include(cmake/Adapter.cmake)
    add_executable(solid-baseline src/baseline.cpp src/profile.cpp)
    target_include_directories(solid-baseline PRIVATE third_party/llama.cpp/vendor "${CMAKE_BINARY_DIR}/generated")
    target_link_libraries(solid-baseline PRIVATE solid_config llama)
    target_compile_definitions(solid-baseline PRIVATE SOLID_UPSTREAM_COMMIT="${SOLID_UPSTREAM_COMMIT}")
endif()

if(SOLID_BUILD_TESTS)
    enable_testing()
    add_executable(solid-config-test tests/config_test.cpp)
    target_link_libraries(solid-config-test PRIVATE solid_config)
    add_test(NAME config COMMAND solid-config-test)
    add_executable(solid-core-test tests/core_test.cpp)
    target_link_libraries(solid-core-test PRIVATE solid_core)
    add_test(NAME kv-and-attention COMMAND solid-core-test)
    add_executable(solid-layout-test tests/layout_test.cpp)
    target_link_libraries(solid-layout-test PRIVATE solid_core)
    add_test(NAME interleaved-layout COMMAND solid-layout-test)
    if(SOLID_BUILD_RUNNER)
        add_executable(solid-graph-test tests/graph_test.cpp)
        target_link_libraries(solid-graph-test PRIVATE solid_core llama)
        add_test(NAME strided-attention-graph COMMAND solid-graph-test)
        add_executable(solid-gather-test tests/gather_test.cpp)
        target_link_libraries(solid-gather-test PRIVATE solid_core llama)
        add_test(NAME gathered-flash-attention COMMAND solid-gather-test)
        add_executable(solid-projection-test tests/projection_test.cpp)
        target_include_directories(solid-projection-test PRIVATE include)
        target_link_libraries(solid-projection-test PRIVATE llama)
        add_test(NAME fused-kv-projection COMMAND solid-projection-test)
        add_executable(solid-profile-test tests/profile_test.cpp src/profile.cpp)
        target_include_directories(solid-profile-test PRIVATE include)
        target_link_libraries(solid-profile-test PRIVATE llama)
        add_test(NAME execution-profile COMMAND solid-profile-test)
        add_test(NAME baseline-help COMMAND solid-baseline --help)
        add_test(NAME baseline-build-info COMMAND solid-baseline --build-info)
        add_test(NAME locked-model-config COMMAND solid-baseline --validate-config
            "${PROJECT_SOURCE_DIR}/configs/qwen2.5-7b-q4_k_m.json")
        foreach(SOLID_MODE offload offload-sparse indram-sparse indram-reference
                           offload-interleaved offload-sparse-interleaved indram-sparse-interleaved)
            add_test(NAME config-${SOLID_MODE} COMMAND solid-baseline --validate-config
                "${PROJECT_SOURCE_DIR}/configs/${SOLID_MODE}.json")
        endforeach()
    endif()
endif()
